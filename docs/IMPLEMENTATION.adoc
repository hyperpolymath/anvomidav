== Overview

This document defines the complete Rust-based implementation architecture for Anvomidav. All components—compiler, runtime, tooling, and visualization—are implemented in Rust, with WASM compilation for browser/mobile deployment.

toc::[]

=== anv-types

Type system implementation with refinement types.

[source,rust]
----
// Types
pub enum Type {
    // Primitives
    Unit, Bool, Int, Float, String,
    // Dimensional
    Time, Duration, Angle, Position, Velocity,
    // Skating
    Edge, Level, Element, Jump, Spin, Step, Lift,
    // Compound
    Tuple(Vec<Type>),
    Record(Vec<(String, Type)>),
    Function { params: Vec<Type>, ret: Box<Type>, effects: Effects },
    // Polymorphic
    Var(TypeVar),
    Forall { vars: Vec<TypeVar>, body: Box<Type> },
    // Refinement
    Refined { base: Box<Type>, predicate: Predicate },
    // Dependent
    Dependent { param: String, param_ty: Box<Type>, body: Box<Type> },
}

// Refinement predicates (for SMT)
pub enum Predicate {
    True,
    False,
    Var(String),
    Int(i64),
    Binary { op: PredOp, lhs: Box<Predicate>, rhs: Box<Predicate> },
    App { func: String, args: Vec<Predicate> },
}

// Type inference
pub struct TypeChecker {
    env: TypeEnv,
    constraints: Vec<Constraint>,
    smt: SmtContext,
}

impl TypeChecker {
    pub fn infer(&mut self, expr: &Expr) -> Result<Type, TypeError>;
    pub fn check(&mut self, expr: &Expr, expected: &Type) -> Result<(), TypeError>;
    pub fn unify(&mut self, t1: &Type, t2: &Type) -> Result<Substitution, TypeError>;
    pub fn solve_refinements(&mut self) -> Result<(), RefinementError>;
}
----
